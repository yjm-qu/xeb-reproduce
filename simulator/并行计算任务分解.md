```python
from multiprocessing import Pool

def run_single_experiment(k):
    """单次实验：k 是种子"""
    circuit = generate_random_circuit(n, m, seed=k)
    bitstrings, psi = run_circuit(circuit, use_noise, N_s)
    return compute_F_XEB(psi, bitstrings, n)

# 并行调用
if __name__ == '__main__':
    with Pool(K_REPEATS) as pool:
        F_list = pool.map(run_single_experiment, range(K_REPEATS))
```
